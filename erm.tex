    \large{1. Empirical Risk Minimization}
  \end{center}
\end{frame}

\begin{frame}
  \frametitle{Empirical Risk Minimization}
  \begin{itemize}
  \item Principle on which many \blue{supervised learning} algorithms are based.
  \item Ingredients:
    \begin{itemize}
    \setlength{\itemsep}{5pt}
    \item \blue{Hypothesis space} \\ \hspace{1em} Which models can we learn?
    \item \blue{Loss / cost / error function} \\ \hspace{1em} How can we quantify the model’s error?
    \item \blue{Optimization algorithm} \\ \hspace{1em} How can we find a model that makes few errors?
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.1 Hypothesis Space}
  \begin{itemize}
  \item \blue{Data:} $\dset = \{(\xvec_1, y_1), (\xvec_2, y_2), \dots, (\xvec_n, y_n)\}$ 
    \begin{itemize}
    \item $n$ observations in $p$ dimensions: $\xvec_i \in  \RR^p$
    \item $n$ labels: $y_i \in \ycal$
    \end{itemize}
    \pause
  \item \red{Hypothesis space:} the set $\fcal$ of \blue{models} that the learning algorithm will consider.
    \pause
  \item \red{Parametric models:} a parameterized family of functions is considered.
    \begin{itemize}
    \item[] \black{Example:} 
      $\fcal = \{ \xvec \mapsto {\color{MyOrange}\alpha_1} x_1^{\beta_1} +
      {\color{MyOrange}\alpha_2} x_2^{\beta_2} + \dots + {\color{MyOrange}\alpha_p} x_p^{\beta_p} \}$ \\
      The goal of learning is to determine $\alpha_1, \dots, \alpha_p, \beta_1, \dots, \beta_p$.
    \end{itemize}
  % \item \red{Non-parametric models}
  %   \begin{itemize}
  %   \item[] \black{Example:} $\fcal = \left\{\xvec \mapsto
  %       1 \text{ if } x_j > 1, 0 \text{ otherwise.} \right\}$ \\
  %     The goal of learning is to determine $j$, which is not a parameter.    
  %   \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.2 Loss Function}
  \begin{itemize}
  \item \blue{Quantify} a model’s error for a given observation.
  \item \red{Loss function:} a function
    $L : \ycal \times \ycal \mapsto \RR$ 
  \item[] $L(y, f(\xvec))$ increases as the error between the predicted value $f(\xvec)$ and the true value $y$ grows.
    \pause
    \begin{itemize}
    \item \red{0/1 loss} for classification:
      \begin{eqnarray*}
        L : & \{0, 1\} \times \{0, 1\} \rightarrow \RR \\ \nonumber
            & (y, f(\xvec)) \mapsto
              \begin{cases}
                0 & \text{if } f(\xvec) = y \\ \nonumber
                1 & \text{otherwise.}
              \end{cases}                
      \end{eqnarray*}
      \pause
    \item \red{Quadratic loss} for regression:
      \begin{eqnarray*}
        L : & \RR \times \RR \rightarrow \RR \\ \nonumber
            & (y, f(\xvec)) \mapsto (y - f(\xvec))^2
      \end{eqnarray*}      
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.3 Empirical Risk}
  \begin{itemize}
  \item \red{Risk:}
    \begin{itemize}
    \item Expected model error over all possible data 
      \begin{mdframed}[hidealllines=true, backgroundcolor=MyLightGrey,
                       fontcolor=MyDarkGrey, leftmargin=-5pt,
                       innerleftmargin=5pt, skipabove=5pt]
      \item Formally: the expectation of the loss function 
        \[\rcal(f) = \EE [L(Y, f(X))]\]
      \item[]
        \begin{itemize}
        \item $(\xvec_1, y_1), (\xvec_2, y_2), \dots, (\xvec_n, y_n)$
          are samples drawn from $(X, Y)$
        \item $X$ is a real random vector of dimension $p$; $Y$ a random variable
          in $\ycal$
        \end{itemize}
      \end{mdframed}
    \end{itemize}
  \item \red{Empirical risk:} average of the loss function over the data
    \[\rcal_{\text{emp}}(f) = \frac1n \sum_{i=1}^n L(y_i, f(\xvec_i)) \]
  \end{itemize}
\end{frame}


\begin{frame}
  \frametitle{1.4 Empirical Risk Minimization}
  \begin{itemize}
  \item \blue{Learning a model} = \blue{finding} a model within the \blue{hypothesis space}
    that \blue{minimizes} the \blue{empirical risk}
    \[ \hat{f} = \argmin_{f \in \fcal} \frac1n \sum_{i=1}^n L(y_i, f(\xvec_i)) \]
  \item It is therefore an \blue{optimization} problem.
  \item Depending on the hypothesis space, the loss function, and the chosen
    optimization procedure:
    \begin{itemize}
    \item The solution may be unique (but not necessarily)
    \item The solution may be explicit (rarely)
    \item The solution may be reached with the desired precision (sometimes)
    \item The solution may only be found using heuristics 
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.5 Maximum Likelihood Estimation}
  \begin{itemize}
  \item In the case of a \blue{parametric regression model:}
    $\fcal = \{f_{\thetavec}  : \RR^p \rightarrow \RR ; \thetavec \in \RR^d \}$
  \item Using the \blue{quadratic loss:} $L(y, f(\xvec)) = (y - f(\xvec))^2 $
  \item Assuming:
    \begin{itemize}
    \item $(\xvec_1, y_1), (\xvec_2, y_2), \dots, (\xvec_n, y_n)$ are samples
      drawn from $(X, Y)$
    \item $X$ is a real random vector of dimension $p$; $Y$ is a real-valued random variable
    \item There exists $f^* : \RR^p \rightarrow \RR$ such that $Y = f^*(X) + \epsilon$
    \item $\epsilon \sim \ncal(0, \sigma^2)$ (Gaussian noise)
    \end{itemize}
  \item Then: the maximum likelihood estimate of $\thetavec$ is
    \blue{equivalent} to minimizing the empirical risk.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.1 Univariate Linear Regression}
  \begin{itemize}
  \item \blue{Hypothesis space:}  $\fcal = \{x \mapsto a x + b ; (a, b) \in \RR^2\}$ 
    \pause
  \item \blue{Loss function:} quadratic loss $L(y, f(x)) = (y - f(x))^2$
    \pause
  \item \blue{Empirical risk minimization:}
    \[ (\hat{a}, \hat{b}) = \argmin_{(a,b) \in \RR^2} \frac1n \sum_{i=1}^n (y_i - (a x_i + b))^2 \]
    \pause
  \item[] \red{Ordinary least squares} — known since Legendre and Gauss. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Solution}
  \begin{itemize}
  \item \blue{By setting the partial derivatives to zero:} a system of two equations with two unknowns.
  \item[] Explicit solution:
    \begin{itemize}
    \item If $\sum_{i=1}^n x_i^2 \neq \left(\sum_{i=1}^n x_i\right)^2,$ there is a \blue{unique solution:}
      \[ a = \frac{\sum_{i=1}^n x_i y_i - \frac1n \sum_{i=1}^n x_i \sum_{i=1}^n y_i}{
          \sum_{i=1}^n x_i^2 - \frac1n \left(\sum_{i=1}^n x_i\right)^2}  \hspace{2em}
        b = \frac1n \sum_{i=1}^n y_i - \frac{a}{n} \sum_{i=1}^n x_i\]
      \pause
    \item Otherwise, the system is \blue{underdetermined} and there are infinitely many solutions.
    \end{itemize}
  \pause
  \item \blue{By gradient descent algorithm}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.2 Multivariate Linear Regression}
  \begin{itemize}
  \item Also known as \blue{multiple linear regression}
  \item \blue{Data:} $\dset = \{(\xvec_1, y_1), (\xvec_2, y_2), \dots, (\xvec_n, y_n)\}$
    \begin{itemize}
    \item $n$ observations in $\RR^p$
    \item $n$ labels: $y_i \in \RR$
    \end{itemize}
  \item \blue{Hypothesis space:}
    \[\fcal = \left\{ \xvec \mapsto \beta_0 + \sum_{j=1}^p \beta_j x_j \, ;  \betavec \in \RR^{p+1} \right\}\]
    \pause
    \vspace{-1em}
  \item \blue{Loss function:} quadratic loss $L(y, f(\xvec)) = (y - f(\xvec))^2$
    \pause
  \item \blue{Empirical risk minimization:}
    \[ \hat{\betavec} = \argmin_{\betavec \in \RR^{p+1}} \frac1n \sum_{i=1}^n \left(y_i - \left(\beta_0 + \sum_{j=1}^p \beta_j x_j\right)\right)^2 \]
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Matrix Form}
  \small
    \[ \hat{\betavec} = \argmin_{\betavec \in \RR^{p+1}} \frac1n \sum_{i=1}^n \left(y_i - \left(\beta_0 + \sum_{j=1}^p \beta_j x_j\right)\right)^2 \]
  \begin{itemize}
  \item The transformation 
    $\xvec \leftarrow (1, x_1, \dots, x_p)$ allows us to write $f(\xvec) = \langle \betavec, \xvec \rangle$ 
  \item Let $X \in \RR^{n \times (p+1)}$ be the data matrix with rows $X_{i.} = \xvec_i$, and \\
    $\yvec \in \RR^n$ the vector of labels:
    \[\sum_{i=1}^n \left(y_i - \left(\beta_0 + \sum_{j=1}^p \beta_j x_j\right)\right)^2 =
    \left(\yvec - X \betavec \right)^\top \left(\yvec - X \betavec \right)\]
  \item The problem becomes:
    \[\argmin_{\betavec \in \RR^{p+1}} \frac1n 
    \left(\yvec - X \betavec \right)^\top \left(\yvec - X \betavec \right)\]
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Solution}
    \small
  \begin{flalign*}
    \text{Minimize } J & : \RR^{p+1} \rightarrow \RR \\
    & \betavec \mapsto \left(\yvec - X \betavec \right)^\top
    \left(\yvec - X \betavec \right) 
  \end{flalign*}
  \begin{itemize}
  \item \blue{Convex optimization} problem
    \pause
  \item \blue{Gradient} of $J$: $\nabla J (\betavec) = - 2 X^\top (\yvec - X \betavec)$
  \item By \blue{setting the gradient to zero}, we obtain:
    \[
      X^\top X \betavec = X^\top \yvec
    \]
    \vspace{-2em}
    \pause
  \item Solution:
    \begin{itemize}
    \item If \blue{$X^\top X$ is invertible}, then $\betavec^* = \left(X^\top X\right)^{-1} X^\top \yvec$
    \item Otherwise, the system is underdetermined and there are infinitely many solutions.
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Invertibility of $X^\top X$}
  \begin{itemize}
  \item $X^\top X$ is invertible if and only if $X$ has full column rank.
    \pause
    \begin{itemize}
    \item This is not the case if $p+1 > n$
    \item[] i.e., if there are \blue{more variables than observations}
    \vspace{1em}
    \item This is not the case if some columns are collinear
    \item[] i.e., if some \blue{variables are highly correlated}
    \end{itemize}
    \pause
  \item Inverting a square matrix of size $p$ is an operation of complexity $\mathcal{O}(p^3)$
  \item[] The \blue{gradient descent algorithm} is generally used even when $X^\top X$ is invertible. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression}
  \begin{itemize}
  \item \blue{Data:} $\dset = \{(\xvec_1, y_1), (\xvec_2, y_2), \dots, (\xvec_n, y_n)\}$
    \begin{itemize}
    \item $n$ observations in $\RR^p$
    \item $n$ labels: $y_i \in \{0, 1\}$ — \blue{binary classification}
    \end{itemize}
  \pause
  \item \blue{Hypothesis space:} Linear functions do not make sense to predict a binary value.
    \pause
    \begin{itemize}
    \item $f$ will model the \blue{probability of belonging to the positive class}
    \pause
    \item The \red{logistic function} $\sigma$ transforms a real value into a “probability”:
    \begin{center}
    \includegraphics[width=0.38\textwidth]{figures/logistic}
    \end{center}
    \item $\fcal = \left\{\xvec \mapsto \sigma\left(\beta_0 + \langle \betavec, \xvec \rangle\right) ;  \betavec \in \RR^{p+1}\right\}$
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression}
  \begin{itemize}
  \item Loss function: \red{cross-entropy}
    \begin{flalign*}
      L :  \{0, 1\} \times ]0, 1[ & \rightarrow \RR \\ \nonumber
          (y, f(\xvec)) & \mapsto -y \log f(\xvec) - (1-y) \log (1 - f(\xvec)) 
        \end{flalign*}
        \begin{center}
          \includegraphics[width=0.8\textwidth]{figures/cross_entropy}
        \end{center}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression}
  \begin{itemize}
  \item Loss function: \red{cross-entropy}
    \begin{flalign*}
      L :  \{0, 1\} \times ]0, 1[ & \rightarrow \RR \\ \nonumber
          (y, f(\xvec)) & \mapsto -y \log f(\xvec) - (1-y) \log (1 - f(\xvec))
        \end{flalign*}
  \item[] If $y=1$, then $L = -\log f(\xvec)$, and the loss is minimized when $f(\xvec) \rightarrow 1$. 
  \item[] If $y=0$, then $L = -\log(1-f(\xvec))$, and the loss is minimized when $f(\xvec) \rightarrow 0$. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression}
  \begin{itemize}
  \item \blue{Hypothesis space:}
    $\fcal = \{\xvec \mapsto \sigma\left(\beta_0 + \langle \betavec, \xvec
      \rangle\right) ; \betavec \in \RR^{p+1}\}$
  \item \blue{Loss function:}
    $L : (y, f(\xvec))  \mapsto -y \log f(\xvec) - (1-y) \log (1 - f(\xvec))$
  \item \blue{Empirical risk minimization:}
    \footnotesize
    \[\argmin_{\betavec \in \RR^{p+1}} \frac1n \sum_{i=1}^n
      -y_i \log \sigma\left(\beta_0 + \langle \betavec, \xvec_i \rangle\right)
      - (1-y_i) \log \left(1 - \sigma\left(\beta_0 + \langle \betavec, \xvec_i \rangle\right)\right)
    \]
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression — Solution}
  \begin{itemize}
  \item \blue{Minimize}
    {\footnotesize \[J : \betavec \mapsto \sum_{i=1}^n -y_i \log \sigma\left(\beta_0 + \langle
      \betavec, \xvec_i \rangle\right) - (1-y_i) \log \left(1 -
      \sigma\left(\beta_0 + \langle \betavec, \xvec_i \rangle\right)\right)\]}
  \pause
  \item $J$ is a \blue{convex} function of $\betavec$
  \pause
  \item Using $\sigma^\prime (u) = \sigma(u) (1-\sigma(u)),$
    \[
      \nabla J(\betavec) = \sum_{i=1}^n \left( y_i - \frac1{1 + \exp - \left(\beta_0 +
            \langle \betavec, \xvec_i \rangle \right)} \right)
    \]
  \item[$\Rightarrow$] No closed-form solution.
    \pause
  \item Therefore, we systematically use the \blue{gradient descent algorithm.}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression — Interpretation}
  \begin{itemize}
  \item The \blue{regression coefficient}
    $\beta_j$ can be interpreted as the contribution of the corresponding variable ($x_j$) to the model
  \item Provided that the variables are \blue{standardized} to the same scale:
    \[
      x_j \leftarrow \frac{x_j - \overline{x_j}}{\sigma_j}
    \]
    \begin{itemize}
    \item $\overline{x_j}$: mean of $x_j$ over the dataset
    \item $\sigma_j$: standard deviation of $x_j$ over the dataset
    \end{itemize}
  \item Note: $\overline{x_j}$ and $\sigma_j$ must be computed on the \blue{training set only}, to avoid data leakage into the test set.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression — Cross-Entropy and Maximum Likelihood}
  \begin{itemize}
    \item Let $P(y_i\,|\,\xvec_i; \betavec)$ be the probability of predicting class $y_i$ (i.e., the correct class). 
  \item The probability that all predictions are correct is the joint probability of $N$ correct predictions:  
  \begin{equation*}
    \prod_{i=1}^N P(y_i | \xvec_i; \betavec)
  \end{equation*}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression — Cross-Entropy and Maximum Likelihood}
  \begin{itemize}
    \item Maximum likelihood approach: 
    \begin{eqnarray*}
    \betavec^{\ast} &=& \argmax_{\betavec} \prod_{i=1}^N P(y_i | \xvec_i; \betavec) \\
                  &=& \argmax _{\betavec} \sum_{i=1}^N\log P(y_i | \xvec_i; \betavec)\\
                  &=& \argmin_{\betavec} - \sum_{i=1}^N\log P(y_i | \xvec_i; \betavec)
    \end{eqnarray*}    
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Logistic Regression — Cross-Entropy and Maximum Likelihood}
  \begin{itemize}
    \item Assuming we predict the posterior probability, we can write: 
    \begin{equation*}
    \log P(y_i|(\xvec_i; \betavec)) = y_i \log (f(\xvec_i; \betavec)) + (1-y_i)\log (1-f(\xvec_i; \betavec))
    \end{equation*}
    and thus: 
    \begin{equation*}
    \betavec^{\ast} = \argmin_{\betavec} - \sum_{i=1}^N [y_i \log (f(\xvec_i; \betavec)) + (1-y_i)\log (1-f(\xvec_i; \betavec))]
    \end{equation*}
    \item We see that minimizing cross-entropy is equivalent to maximizing likelihood.
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{2.4 Linear Discriminant Analysis — Bayesian Rule 1/4}
  \begin{itemize}
  \item Let $X$ and $Y$ be two random variables, with marginal probabilities $P(X)$ and $P(Y)$, joint probability $P(X,Y)$, and conditional probability $P(X\,|\,Y)$ — the \red{probability of $X$ given $Y$}.
    \begin{itemize}
    \item The \red{joint probability} can be written as:
    \begin{equation}
      P(X,Y) = P(X\,|\,Y)P(Y)
    \end{equation}
    \item The \red{marginal probability} is the sum of the joint probabilities over all values of $Y$:
    \begin{equation}
      P(X) = \sum_Y P(X,Y) = \sum_Y P(X|Y)P(Y)
    \end{equation}
  \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{2.4 Linear Discriminant Analysis — Bayesian Rule 2/4}
\begin{itemize}
\item Since $P(X,Y)=P(Y,X)$, we directly obtain Bayes’ theorem:
\begin{equation}
  P(Y \, | \, X) = \frac{P(X \, | \, Y)P(Y)}{P(X)}
\end{equation}
\item Here $X$ and $Y$ are discrete variables, but a nearly identical rule can be defined for continuous variables by replacing probabilities with probability densities.
\end{itemize}
\end{frame}

\begin{frame}{2.4 Linear Discriminant Analysis — Bayesian Rule 3/4}
\begin{itemize}
  \item \red{Posterior probability:} Let $\xvec \in \mathbb{R}^P$ have a probability density $p(\xvec)$ and $y \in \{1, \ldots, K\}$ be a discrete variable. The posterior probability $P(y=k\,|\,\xvec)$ is:
  \begin{equation}
    P(y=k \, | \, \xvec) = \frac{p(\xvec \, | \, y=k) P(y=k)}{p(\xvec)}
  \end{equation}
  \item \red{Bayesian classification rule:} \emph{Choose the class that maximizes} $P(y=k|\xvec)$:
  \begin{equation}
    \hat{y}(\xvec) = \arg\max_k P(y=k \, | \, \xvec) = \arg\max_k p(\xvec\,|\,y=k)P(y=k)
  \end{equation}
\end{itemize}
\end{frame}

\begin{frame}{2.4 Linear Discriminant Analysis — Bayesian Rule 4/4}
\begin{itemize}
  \item The Bayesian classification rule simply states that the best class to choose is the one with the highest probability given the observation. 
  \item This probability can be estimated from the class-conditional distributions of the observations and the \emph{prior} probabilities. 
  \item This rule is important because it defines the best possible classifier, provided the class-conditional distributions $p(\xvec\,|\,y=k)$ and the priors $P(y=k)$ are known.
  \item Since these distributions are unknown, they must be estimated from the training data. 
\end{itemize}
\end{frame}

\begin{frame}{2.4 Linear Discriminant Analysis — Binary Case}
\begin{itemize}
\item Assume we are solving a binary classification problem, i.e. $y \in \{1, 2\}$. The Bayesian rule then states that we should choose class 1 if: 
\begin{equation*}
\frac{p(\xvec|y=1)P(y=1)}{p(\xvec|y=2)P(y=2)} > 1
\end{equation*}
\item Taking the $log$ of both sides yields:
\begin{equation}\label{equ:lda:condition}
\log{ \frac{p(\xvec|y=1)P(y=1)}{p(\xvec|y=2)P(y=2)} } > 0
\end{equation}
\end{itemize}
\end{frame}

\begin{frame}{2.4 Linear Discriminant Analysis — Normality Assumption}
\begin{figure}[htb]
\includegraphics[width=0.5\textwidth]{figures/LDA.pdf}
\end{figure}
\begin{itemize}
\item We assume that the features are normally distributed within each class:
\begin{equation}\label{equ:lda:normal}
p(\xvec|y=k)=\frac{1}{(2\pi)^{\frac{p}{2}}|\Sigma_k|^{\frac{1}{2}}}e^{-\frac{1}{2}(\xvec-\mu_k)^T\Sigma_k^{-1}(\xvec-\mu_k)}
\end{equation}
\end{itemize}
\end{frame}

\begin{frame}{2.4 Linear Discriminant Analysis}
\frametitle{Linear Discriminant Analysis}
\begin{itemize}
\item Using expressions (\ref{equ:lda:normal}) and (\ref{equ:lda:condition}) and assuming identical covariance matrices across classes, i.e.
$\Sigma_1 = \Sigma_2 = \Sigma$, we obtain the \textbf{Linear Discriminant Analysis (LDA)}:
\begin{equation*}
K + \xvec^T\Sigma^{-1}(\mu_1-\mu_2) - \frac{1}{2}(\mu_1-\mu_2)^T\Sigma^{-1}(\mu_1+\mu_2) > 0
\end{equation*}
where $K=\log{\frac{P(y=1)}{P(y=2)}} + \frac{1}{2}\log{\frac{|\Sigma_2|}{|\Sigma_1|}}$
\item We observe that: 
\begin{itemize}
  \item It is a \blue{linear classifier.} 
  \item The expression involves inverting a covariance matrix (which can be unstable when features are correlated).

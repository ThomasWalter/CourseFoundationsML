\begin{frame}
  \frametitle{1.1 Arbres de décision}
  \begin{itemize}
    \item Méthode hiérarchique
    \item Série successive de tests conditionnels.
    \item Souvent utilisé en dehors de la classification automatique : 
    \begin{itemize}
      \item aide à la décision
      \item formalisation d'un raisonnement
    \end{itemize}
  \end{itemize}
\end{frame}


\begin{frame}
  \frametitle{1.1 Arbres de décision}
  \begin{center}
    \includegraphics[width=.7\textwidth]{figures/arbre}
  \end{center}
  \begin{itemize}
    \item Variables \blue{continues :} tests = \blue{seuillage}
    \item Variables \blue{catégoriques :} tests = \blue{valeur}
    \item Chaque arbre de décision correspond à une \blue{partition} de l'espace des descripteurs.
    \item Feuille = étiquette
  \end{itemize}
\end{frame}


\begin{frame}
  \frametitle{1.2 Avantages}
  \begin{itemize}
  \item Gère naturellement :
    \begin{itemize}
    \item Classification \blue{multi-classe}
    \item Plusieurs \blue{types de variables} (continues, binaires, catégoriques)
    \item Plusieurs régions de l'espace peuvent avoir la même étiquette    
  \end{itemize}
  \item \blue{Interprétabilité}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.3 Construction}
  \begin{itemize}
    \item \red{CART :} Classification and regression trees
    \item A chaque étape, une région est divisée en deux, selon une \blue{variable séparatrice} $x_j$ (et un \blue{point de séparation} $s$ si $x_j$ continue):
      \begin{equation*}
        R_l(j, s) = \{\xvec : x_j < s\} ; \hspace{2em} R_r(j, s) = \{\xvec : x_j \geq s\}.
      \end{equation*}
      % \item Ce choix est fait pour optimiser un critère, calculé sur les régions qui résultent de ce seuillage. 
    \item Approche \blue{gloutonne} : on fait ce choix pour optimiser l'\red{erreur} faite à ce moment là
    \item  Étiquette d'une feuille = étiquette \blue{majoritaire} (classification) ou \blue{moyenne} (régression) des points du jeu d'entraînement dans cette feuille
    \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Critère de séparation pour la régression}
  \begin{itemize}
    % \item Un suppose qu'un split $(j,s)$ génère deux régions $R_1$ et $R_2$. 
  \item Moyenne des étiquettes dans chaque région : $\overline{y_l}$ et $\overline{y_r}$
  \item Critère à minimiser par $(j,s)$: \blue{erreur quadratique moyenne}
    \begin{equation*}
    (j,s)^{\ast} = \argmin_{j,s}\sum_{\xvec^i \in R_l} (y^i-\overline{y_l})^2 + \sum_{\xvec^i \in R_r}(y^i-\overline{y_r})^2
    \end{equation*}
    % où $\overline{y_1}, \overline{y_2}, R_1, R_2$ dépendent de $(j,s)$. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Critère de séparation pour la classification}
  \begin{itemize}
    \item Critère d'\red{impureté de Gini} d'une région $R$ : 
    \begin{equation*}
    \text{Imp}(R) = \sum_{c=1}^{C}p_c(R)(1-p_c(R))
    \end{equation*}
    avec $p_c(R) = \frac1{\lvert R \rvert} \sum_{i : \xvec^i \in R} \indic_{y^i=c}$ la probabilité (proportion) de la classe $c$ dans $R$.     
    \item Critère à minimiser par $(j,s)$:
    \begin{equation*}
    (j,s)^{\ast} = \argmin_{j,s}\frac{\lvert R_l \rvert}{\lvert R \rvert}\text{Imp}(R_l) + \frac{\lvert R_r \rvert}{\lvert R \rvert}\text{Imp}(R_r)
    \end{equation*}
    % avec $|\cdot|$ le cardinal et $R_1, R_2$ des fonctions de $(j,s)$. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Critère d'arrêt}
  \begin{itemize}
  \item Une étiquette par feuille : risque de \blue{sur-apprentissage}
  \item Critères d'\blue{arrêt} :
    \begin{itemize}
    \item Profondeur 
    \item Nombre d'observations par feuille
    \end{itemize}
  \item \blue{Élagage} : regrouper séquentiellement des feuilles jusqu'à minimiser la \blue{complexité} de l'arbre
      \begin{equation*}
        C_\lambda(T) = \sum_{r=1}^{|T|} \lvert R_r \rvert \text{Imp}(R_r) + \lambda \lvert T \rvert.
  \end{equation*}
  \end{itemize}
\end{frame}
  

\begin{frame}[t]
  \frametitle{Construction d'un arbre}
  \begin{itemize}
  \item Exemple :
    \begin{itemize}
    \item 4 petites pommes jaunes, 5 grosses pommes vertes, 3 bananes, 6 mini-concombres
    \item attributs : vert/jaune, rond/pas rond, gros/petit

  \end{itemize}

\begin{frame}
  \frametitle{1.4 Limitations}
  \begin{itemize}
    \item \blue{Temps de calcul} potentiellement long
    \item Grand risque de \blue{sur-apprentissage}
    \item Généralement \blue{peu performants}, encore plus en régression
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.2 Bagging}
  \begin{itemize}
    \item On considère seulement un \blue{sous-ensemble} de $\dset$ (e.g. $80\%$ ou par échantillonage bootstrap (avec remise)). 
    \item On entraîne un arbre de décision. 
    \item On reitère cette procédure pour obtenir un ensemble d'arbres. 
    \item Le résultat est obtenu par vote majoritaire (classification) ou en moyennant (régression).  
    \item Moyenner permet d'avoir des étiquettes précises même en régression
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 Forêts aléatoires}
  \begin{itemize}
    \item En plus du bagging, on peut à chaque noeud tirer aléatoirement des variables parmi lesquels la variable optimale est trouvée. 
    \item Le résultat : \blue{forêts aléatoires}. 
    \item Les arbres sont plus différents et couvrent «~différent aspects~» des données. 
    \item En pratique : 
    \begin{itemize}
      \item généralement bons résultats
      \item applicable pour $p \gg n$
      \item Régularisation par «~ensembling~»
    \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}
  \frametitle{2.4 Boosting}
  \begin{itemize}
  \item À l'itération $m$
    \begin{itemize}
    \item Apprendre le modèle $f_m$ qui minimise l'erreur empirique de
      \[
        F_m = \sum_{l=1}^m \alpha_l f_l = F_{m-1} + \alpha_m f_m 
      \]
    \item L'erreur pour $(\xvec_i, y_i)$ est pondérée de sorte à donner plus
      d'importances aux exemples pour lesquels $F_{m-1}$ se trompe.
    \end{itemize}
  \item \blue{AdaBoost} (Schapire \& Freund 1997) : erreur exponentielle, $f_m$ est un
    arbre de décision de profondeur 1 (decision stump)
  \item \blue{Gradient Boosting} (Friedman 2001) : forme générale
  \end{itemize}
\end{frame}

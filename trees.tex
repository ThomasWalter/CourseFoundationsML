\begin{frame}
  \frametitle{1.2 How to build a tree}
  \begin{itemize}
    \item \red{CART :} Classification And Regression Trees
    \item At each step, split a region in two, according to \blue{separating variable} $x_j$ (and \blue{separating point} $s$ if $x_j$ continuous):
      \begin{equation*}
        R_l(j, s) = \{\xvec : x_j < s\} ; \hspace{2em} R_r(j, s) = \{\xvec : x_j \geq s\}.
      \end{equation*}
      % \item Ce choix est fait pour optimiser un critère, calculé sur les régions qui résultent de ce seuillage. 
    \item \blue{Greedy} approach: chose $j$ (and $s$) that \red{minimize error} at that stage
    \item  Leaf label = \blue{majority vote} (classification) or \blue{average} (regression) labels of all training points that belong to that leaf
    \end{itemize}
\end{frame}

\begin{frame}[t]
  \frametitle{1.2 How to build a tree}
  \black{Separating criteria}
  \begin{itemize}
  \item For \blue{regression:} \red{mean squared error}%Critère à minimiser par $(j,s)$: \blue{erreur quadratique moyenne}
    \begin{equation*}
      (j,s)^{\ast} = \argmin_{j,s}\sum_{\xvec^i \in R_l} (y^i-\overline{y_l})^2 + \sum_{\xvec^i \in R_r}(y^i-\overline{y_r})^2
    \end{equation*}
    \begin{itemize}
      \setlength{\itemsep}{3pt}      
    \item $(j, s)$ creates  $R_l$ (where $x_j \leq s$) and $R_r$ (where $x_j > s$)
    \item $\overline{y_l}$: average label of samples in $R_l$ 
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[t]
  \frametitle{1.2 How to build a tree}
  \black{Separating criteria}
  \begin{itemize}
  \item For \blue{classification:} \red{Gini impurity}
    \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item \red{Gini impurity} of a region $R$:
      \begin{equation*} 
        \text{Imp}(R) = \sum_{c=1}^{C}p_c(R)(1-p_c(R))
      \end{equation*}
      \begin{itemize}
      \item[] $p_c(R) = \frac1{\lvert R \rvert} \sum_{i : \xvec^i \in R} \indic_{y^i=c}$ (proportion of samples of class $c$ in $R$).     
      \end{itemize}
    \item Overall
      \begin{equation*}
        (j,s)^{\ast} = \argmin_{j,s}\frac{\lvert R_l \rvert}{\lvert R \rvert}\text{Imp}(R_l) + \frac{\lvert R_r \rvert}{\lvert R \rvert}\text{Imp}(R_r)
      \end{equation*}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[t]
  \frametitle{1.2 How to build a tree}
  \black{Stopping criterion}
  \begin{itemize}
  \item One sample per leaf: risk of \blue{overfitting}
  \item When to stop growing a tree:
    \begin{itemize}
      \setlength{\itemsep}{3pt}
    \item When a preset maximum \blue{depth} is reached
    \item When a preset minimum \blue{number of samples per leave} is reached
    \end{itemize}
  \item \blue{Pruning:} sequentially group leaves so as to minimize \red{tree complexity} 
      \begin{equation*}
        C_\lambda(T) = \sum_{r=1}^{|T|} \lvert R_r \rvert \text{Imp}(R_r) + \lambda \lvert T \rvert.
  \end{equation*}
  \end{itemize}
\end{frame}

\begin{frame}[t]
  \frametitle{1.3 Limitations of decision trees}
  \begin{itemize}
  \item \blue{Computational complexity:} 
    \begin{itemize}
    \item[] For each node: evaluate \blue{all variables} and \blue{all possible thresholds}
    \end{itemize}
    \item \blue{Heuristic} optimization
    \item Often \blue{perform poorly}, even more so for regression problems
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.4 Boosting}
  \begin{itemize}
  \item Méthode \blue{séquentielle}    
  \item À l'itération $m$
    \begin{itemize}
    \item Apprendre le modèle $f_m$ qui minimise l'erreur empirique de
      \[
        F_m = \sum_{l=1}^m \alpha_l f_l = F_{m-1} + \alpha_m f_m 
      \]
    \item L'erreur pour $(\xvec_i, y_i)$ est pondérée de sorte à donner plus
      d'importances aux exemples pour lesquels $F_{m-1}$ se trompe.
    \end{itemize}
  \item \red{AdaBoost} (Schapire \& Freund 1997) : erreur exponentielle, $f_m$ est un
    arbre de décision de profondeur 1 (\red{decision stump})
  \item \red{Gradient Boosting} (Friedman 2001) : forme générale
  \end{itemize}
\end{frame}

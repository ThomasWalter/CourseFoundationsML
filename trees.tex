\begin{frame}
  \frametitle{1.2 Construction}
  \begin{itemize}
    \item \red{CART :} Classification And Regression Trees
    \item A chaque étape, une région est divisée en deux, selon une \blue{variable séparatrice} $x_j$ (et un \blue{point de séparation} $s$ si $x_j$ continue):
      \begin{equation*}
        R_l(j, s) = \{\xvec : x_j < s\} ; \hspace{2em} R_r(j, s) = \{\xvec : x_j \geq s\}.
      \end{equation*}
      % \item Ce choix est fait pour optimiser un critère, calculé sur les régions qui résultent de ce seuillage. 
    \item Approche \blue{gloutonne} : on fait ce choix pour optimiser l'\red{erreur} faite à ce moment là
    \item  Étiquette d'une feuille = étiquette \blue{majoritaire} (classification) ou \blue{moyenne} (régression) des points du jeu d'entraînement dans cette feuille
    \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.2 Construction}
  \black{Critère de séparation}
  \begin{itemize}
  \item Pour la \blue{régression :} \red{erreur quadratique moyenne}
    \begin{equation*}
      (j,s)^{\ast} = \argmin_{j,s}\sum_{\xvec^i \in R_l} (y^i-\overline{y_l})^2 + \sum_{\xvec^i \in R_r}(y^i-\overline{y_r})^2
    \end{equation*}
    \begin{itemize}
      \setlength{\itemsep}{3pt}      
    \item $(j,s)$ génère deux régions $R_l$ (où $x_j \leq s$) et $R_r$ (où $x_j > s$)
    \item $\overline{y_l}$ : moyenne des étiquettes dans $R_l$ 
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.2 Construction}
  \black{Critère de séparation}
  \begin{itemize}
  \item Pour la \blue{classification :} \red{impureté de Gini}
    \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item \red{Impureté de Gini} d'une région $R$:
      \begin{equation*} 
        \text{Imp}(R) = \sum_{c=1}^{C}p_c(R)(1-p_c(R))
      \end{equation*}
      \begin{itemize}
      \item[] $p_c(R) = \frac1{\lvert R \rvert} \sum_{i : \xvec^i \in R} \indic_{y^i=c}$ (proportion de la classe $c$ dans $R$).     
      \end{itemize}
    \item Globalement :
      \begin{equation*}
        (j,s)^{\ast} = \argmin_{j,s}\frac{\lvert R_l \rvert}{\lvert R \rvert}\text{Imp}(R_l) + \frac{\lvert R_r \rvert}{\lvert R \rvert}\text{Imp}(R_r)
      \end{equation*}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.2 Construction}
  \black{Critère d'arrêt}
  \begin{itemize}
  \item Une étiquette par feuille : risque de \blue{sur-apprentissage}
  \item Critères d'\blue{arrêt} :
    \begin{itemize}
    \item Quand la \blue{profondeur} maximale est atteinte
    \item Quand le \blue{nombre d'observations par feuille} minmal est atteint
    \end{itemize}
  \item \blue{Élagage} : regrouper séquentiellement des feuilles jusqu'à minimiser la \blue{complexité} de l'arbre
      \begin{equation*}
        C_\lambda(T) = \sum_{r=1}^{|T|} \lvert R_r \rvert \text{Imp}(R_r) + \lambda \lvert T \rvert.
  \end{equation*}
  \end{itemize}
\end{frame}

\begin{frame}[t]
  \frametitle{1.3 Limites}
  \begin{itemize}
  \item \blue{Temps de calcul} potentiellement long:
    \begin{itemize}
    \item À chaque nœud : évaluer \blue{toutes les variables} et \blue{tous les seuils}
    \item Pour une variable réelle, autant de seuils possibles que de nombres de valeurs différentes prises par cette variable (-1)
    \end{itemize}
    \item Optimisation \blue{heuristique}
    \item Généralement \blue{peu performants}, encore plus en régression
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.4 Boosting}
  \begin{itemize}
  \item Méthode \blue{séquentielle}    
  \item À l'itération $m$
    \begin{itemize}
    \item Apprendre le modèle $f_m$ qui minimise l'erreur empirique de
      \[
        F_m = \sum_{l=1}^m \alpha_l f_l = F_{m-1} + \alpha_m f_m 
      \]
    \item L'erreur pour $(\xvec_i, y_i)$ est pondérée de sorte à donner plus
      d'importances aux exemples pour lesquels $F_{m-1}$ se trompe.
    \end{itemize}
  \item \red{AdaBoost} (Schapire \& Freund 1997) :
    \begin{itemize}
    \item erreur exponentielle
    \item $f_m$ est un arbre de décision de profondeur 1 (\red{decision stump})
  \end{itemize}
  \item \red{Gradient Boosting} (Friedman 2001) : forme générale
  \end{itemize}
\end{frame}

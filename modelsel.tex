\begin{frame}
  \frametitle{1.1 Generalization and overfitting}
  \begin{itemize}
  \item[] \blue{Main challenge} of supervised learning:
    \begin{itemize}
    \item It's quite easy to train a model that works well (low prediction error) on the training data
    \item[] Extreme example: learning ``by heart''
      \vspace{.5em}
    \pause
  \item \red{Generalization :} a model's ability to \blue{make good predictions on samples with unknown labels} 
      \vspace{.5em}
    \pause
    \item \red{Overfitting:} when the model performs better on training data than on other data
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Bias-variance tradeoff}
  \begin{itemize}
  \item Hypotheses:
    \begin{itemize}
    \item $\dset = (\xvec^1, y^1), \dots, (\xvec^n, y^n)$ \blue{random sample} of $(X, Y)$
    \item $Y = \Phi(X) + \varepsilon$ \hspace{2em} $\mathbb{E}[\varepsilon] = 0$ and $\text{Var}[\varepsilon] = \sigma^2$
    \end{itemize}
  \item \blue{Learned model} $f$ $\Rightarrow$ prediction $f(\xvec)$
    \begin{itemize}
    \item estimation of $\Phi(\xvec)$
    \item realization of a random variable $F_n$ (that depends on $\dset$)
  \end{itemize}
  \item \blue{Mean squared error} of the estimator :
    \[       \mathbb{E}[(F_n - y)^2] = \mathbb{E}[(F_n - \Phi(\xvec^i) - \varepsilon)^2] = \dots = \text{Bias}(F_n)^2 + \text{Var}(F_n) + \sigma^2 
    \]    
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.3 Cross-validation}
  \begin{itemize}
  \item \red{Cross-validation} 
    \begin{itemize}
    \item uses every sample for training \textit{and} evaluation
    \item gives an average performance (+/- standard error) --- reduced dependency to the choice of one test set
    \end{itemize}
  \item Split the data into K \blue{folds}
    \begin{itemize}
    \item[] In practice, K=5 or K=10 
    \item[] balance between set size and number of experiments
    \end{itemize}
  \item Use each fold in turn as \blue{test set} and the union of the others as \blue{training set}
  \item[$\Rightarrow$] K performance scores $\rightarrow$ \blue{average generalization performance} 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{1.5 Model (regression)}
  \begin{itemize}
  \item Counting \blue{errors} has no meaning
    \pause
  \item \red{Residual sum of squares:} $\text{RSS} = \sum_{i=1}^n \left(y_i - f(\xvec_i) \right)^2$
  \item \red{\tikzmark{ml0}{Root} Mean Squared Error}
    \[\text{RMSE} = \sqrt{\frac1n \sum_{i=1}^n \left(y_i - f(\xvec_i) \right)^2}\]
  \item \red{Relative Squared Error:} $\text{RSE} = \frac{\sum_{i=1}^n \left(y_i - f(\xvec_i) \right)^2}{\sum_{i=1}^n \left(y_i - \overline{y} \right)^2}$ \hspace{1em} $\overline{y} = \sum_{l=1}^n y_l$
  \item \red{\tikzmark{ml1}{Coefficient} of determination} (can be negative despite the name)
    \[
      R^2 = 1 - \text{RSE} = \frac{\sum_{i=1}^n \left( y_i - \overline{y} \right) \left( f(\xvec_i) - \overline{f(\xvec)} \right) }{\sqrt{\sum_{i=1}^n \left(y_i - \overline{y} \right)^2} \sqrt{\sum_{i=1}^n \left(f(\xvec_i) - \overline{f(\xvec)} \right)^2}}
    \]
  \end{itemize}

\begin{frame}
  \frametitle{2.1 Regularization}
  \begin{itemize}
  \item \blue{Empirical risk minimization:} overfitting by design?
    \[ \hat{f} = \argmin_{f \in \fcal} \frac1n \sum_{i=1}^n L(y_i, f(\xvec_i)) \]
    \pause
  \item \red{Regularization:} prevent overfitting by \blue{limiting model complexity}
    \[ \hat{f} = \argmin_{f \in \fcal} \frac1n \sum_{i=1}^n L(y_i, f(\xvec_i))
      {\color{MyOrange} \, + \, \lambda \, \Omega(f) } \hspace{1em} \lambda > 0
    \]
    \begin{itemize}
    \item \red{Model complexity} theoretical notion, see e.g. Vapnik--Chervonenkis
    \item Proxies: number/size of the parameters
    \end{itemize}
  \end{itemize}
\end{frame}

    \begin{itemize}

\begin{frame}
  \frametitle{2.2 Ridge regularization}
  \begin{itemize}
  \item Example 1: \red{Ridge regularization} (or \red{$\ell_2$}, or \red{weight decay})
    \begin{itemize}
    \item For $f$ parametrized by $\theta_1, \theta_2, \dots, \theta_d$
    \item $\Omega(\thetavec) = \ltwonorm{\thetavec}^2 = \sum_{j=1}^d \theta_j^2$ 
    controls \blue{parameters magnitude} 
      \pause
      \vspace{1em}
    \item \red{Ridge regression:}
      \small
      \[\argmin_{\betavec \in \RR^{p+1}} \frac1n 
        \left(\yvec - X \betavec \right)^\top \left(\yvec - X \betavec
        \right) + \lambda \ltwonorm{\betavec}^2 \]
    \item[] Always admits a unique solution
      $\betavec^* = \left(X^\top X + \lambda I_p \right)^{-1} X^\top \yvec$
      % car ajouter une matrice diagonale à coefficients strictement positifs à
      % $X^\top X$ la rend inversible
    \end{itemize}
  \end{itemize}
\end{frame}

    \begin{itemize}

    \item $\Omega(\thetavec) = \lonenorm{\thetavec} = \sum_{j=1}^d \lvert \theta_j \rvert$

    \item \red{Lasso} (Least Absolute Sparse Selection Operator) :

        \left(\yvec - X \betavec \right)^\top \left(\yvec - X \betavec
        \right) + \lambda \lonenorm{\betavec} \]
    \item Solved by gradient descent
    \end{itemize}

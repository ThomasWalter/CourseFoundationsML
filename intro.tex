\begin{frame}
  \frametitle{1.1 Supervised Learning}
  \begin{center}
    \begin{tikzpicture}
      \tikzstyle{every node}=[font=\small]
      \node[text width=5em] at (0.1, 5.5) {Labeled observations};
      \draw[->, draw=MyDarkGrey, thick] (0.9, 5.5) -- (1.9, 5.5);
      \draw [draw=MyDarkGrey, thick] (2, 6) rectangle (4.5, 5)
      node[midway, text width=5em, align=center]{Learning algorithm};
      \draw[->, draw=MyDarkGrey, thick] (4.6, 5.5) -- (5.6, 5.5);
      \node[text width=5em] at (7, 5.5) {Predictive model};
    \end{tikzpicture}
  \end{center}

  \begin{itemize}
  \item \black{Goal:} \red{learn} a \blue{predictive model}
  \pause
  \item \black{Formalization:}
    \begin{itemize}
    \item[] \blue{Data:} 
    \item $n$ observations $\{\xvec_1, \xvec_2, \dots, \xvec_n\}$; $\xvec_i \in \xcal$ \only<2->{$= \RR^p$}
    \item $n$ labels $\{y_1, y_2, \dots, y_n\}$; $y_i \in \ycal$
      \begin{itemize}
      \item<3-> \red{Regression:} $\ycal = \RR$
      \item<4-> \red{Classification:} $\ycal = \{0, 1, \dots, C-1\}$
      \item<5-> \red{Binary classification:} $\ycal = \{0, 1\}$
      \end{itemize}
    \item[] \blue{Output:}
    \item \blue{Model} $f: \xcal \rightarrow \ycal$ such that $f(\xvec_i) \approx y_i$.
    \end{itemize}
  \item<6-> \red{Inference:} given $\xvec \in \xcal$, predict $\hat{y} = f(\xvec)$.
  \end{itemize}
\end{frame}

\begin{frame}

  \begin{itemize}
  \item[] Learn $f : \RR^p \rightarrow \RR^d$ with $d \ll p$ such that

  \item \black{Use cases:}
    \begin{itemize}

    \item Reducing data size in memory
    \item Improving the performance of a supervised algorithm
    \end{itemize}
  \end{itemize}

\begin{frame}
  \frametitle{2.1 Nearest Neighbor Algorithm}
  \begin{overlayarea}{\textwidth}{\textheight}
  \begin{itemize}
  \item<1-> \blue{Data:} $\dset = \{(\xvec_1, y_1), (\xvec_2, y_2), \dots, (\xvec_n, y_n)\}$ 
    \begin{itemize}
    \item $n$ observations in $p$ dimensions: $\xvec_i \in \RR^p$
    \item $n$ labels: $y_i \in \ycal$
    \end{itemize}
  \item<2-> \blue{Model:} associate to $\xvec \in \RR^p$ the label of the example
    in $\dset$ that is closest to $\xvec$ according to the Euclidean distance
  \[ f(\xvec) = y_{\argmin_{i=1, \dots, n} \ltwonorm{\xvec - \xvec_i}} \]
    \only<3>{\centering \includegraphics[width=0.45\textwidth]{figures/nearest_neighbor}}
    \item<4-> Regression, binary classification, multiclass classification.
    \item<5-> \red{No training phase!}
    \end{itemize}
  \end{overlayarea}
\end{frame}

\begin{frame}
  \frametitle{2.2 kNN for Classification}
  \begin{itemize}
  \item The nearest neighbor algorithm is \blue{not very robust} to noise.
  \item Instead of using \blue{the} single nearest neighbor, use the $k$
    closest observations.
  \item \red{Majority vote:} assign to $\xvec$
    \blue{the majority label} among its $k$ nearest neighbors.
    \pause
      \begin{mdframed}[hidealllines=true, backgroundcolor=MyLightGrey,
                       fontcolor=MyDarkGrey, leftmargin=-5pt,
                       innerleftmargin=5pt, skipabove=5pt]
    \[ f(\xvec) = \argmax_{c \in \{0, 1, \dots, C-1\}} \sum_{i : \xvec_i \in \ncal_k(\xvec)} \delta(y_i, c) \]
      \begin{itemize}
      \item $\ncal_k(\xvec)$ is the set of the $k$ nearest neighbors of $\xvec$ in $\dset$
      \item $\delta(y_i, c) = 1$ if $y_i = c$ and $0$ otherwise; $C$ is the number of classes.
      \end{itemize}
    \end{mdframed}
  \pause
  \item For binary classification, an \blue{odd} number of neighbors is used to avoid ties.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{2.3 kNN for Regression}
  \begin{itemize}
  \item \red{Averaging:} assign to $\xvec$ 
    \blue{the mean label value} of its $k$ nearest neighbors
    \[ f(\xvec) = \frac1k \sum_{i : \xvec_i \in \ncal_k(\xvec)} y_i \]
  \end{itemize}
  \pause
    \begin{itemize}
    \item \black{Note:} Models produced by nearest neighbor algorithms are
      \red{non-parametric models:}
      \begin{itemize}
      \item they do not involve learning parameters of an explicit function of
        the variables $x_1, \dots, x_p$
      \item as opposed to \blue{linear models} or \blue{neural networks}
      \item $k$ is an \red{hyperparameter:} the algorithm does not learn it automatically.
      \end{itemize}
    \end{itemize}
\end{frame}

{\setbeamercolor{background canvas}{bg=MyLightGrey}
  \begin{frame}
    \frametitle{2.4 Variants}
    \begin{itemize}
    \item \blue{Epsilon-neighbors:} instead of considering the $k$ nearest
      neighbors, consider all observations within a ball of radius $\epsilon$
      centered on $\xvec$.
    \item \blue{Weighted neighbors:} the contribution of each neighbor is
      weighted by a coefficient inversely proportional to its distance to $\xvec$.
    \item \blue{Other distances:} the algorithm can also be applied using
      other distance or similarity measures:
      \begin{itemize}
      \item Minkowski distance
      \item Cosine similarity
      \item Hamming distance between binary vectors
      \end{itemize}
    \end{itemize}
  \end{frame}}
